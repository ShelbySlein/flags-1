// Searchable text input with auto-suggest datalist
viewof searchQuery = {
  const form = html`
    <div style="margin-bottom: 25px;">
      <label style="font-weight: 600; font-size: 16px; display: block; margin-bottom: 8px;">
        🔍 Search Country:
      </label>
      <input 
        type="text" 
        list="country-suggestions" 
        placeholder="Type a country name (e.g. United States, Japan, France)..." 
        value="United States" 
        style="width: 100%; max-width: 400px; padding: 10px 14px; font-size: 15px; border: 2px solid #ccc; border-radius: 6px; outline: none; box-sizing: border-box;"
      />
      <datalist id="country-suggestions">
        ${country_list.map(c => html`<option value="${c}">`)}
      </datalist>
    </div>
  `;
  const input = form.querySelector("input");
  input.oninput = () => {
    form.value = input.value;
    form.dispatchEvent(new CustomEvent("input"));
  };
  form.value = input.value;
  return form;
}
matchedCountry = {
  const query = (searchQuery || "").trim().toLowerCase();
  if (!query) return country_list[0];
  
  // 1. Exact match
  const exact = country_list.find(c => c.toLowerCase() === query);
  if (exact) return exact;
  
  // 2. Starts with query
  const starts = country_list.find(c => c.toLowerCase().startsWith(query));
  if (starts) return starts;
  
  // 3. Contains query
  const contains = country_list.find(c => c.toLowerCase().includes(query));
  if (contains) return contains;
  
  return country_list[0];
}

topSims = transpose(top_similar_data).filter(d => d.country_1 === matchedCountry)
mostDiff = transpose(most_different_data).filter(d => d.country_1 === matchedCountry)
selfFlag = transpose(self_flags_data).find(d => d.country_1 === matchedCountry)

html`
